   219→   219→function ConsentTab({
   220→   220→  consent,
   221→   221→  participantById,
   222→   222→  onChanged,
   223→   223→}: {
   224→   224→  consent: { id: string; name: string; email: string; department: string; consented: boolean; consentedAt: string | null }[];
   225→   225→  participantById: Map<string, Participant>;
   226→   226→  onChanged: () => void;
   227→   227→}) {
   228→   228→  const [addOpen, setAddOpen] = useState(false);
   229→   229→  const [pendingId, setPendingId] = useState<string | null>(null);
   230→   230→
   231→   231→  const setConsent = async (id: string, consented: boolean) => {
   232→   232→    setPendingId(id);
   233→   233→    try {
   234→   234→      await api<{ participant: Participant }>(`/api/participants/${id}`, {
   235→   235→        method: "PATCH",
   236→   236→        body: JSON.stringify({ consented }),
   237→   237→      });
   238→   238→      toast.success(consented ? "Consent recorded" : "Consent withdrawn");
   239→   239→      onChanged();
   240→   240→    } catch (err) {
   241→   241→      toast.error(apiErrorMessage(err));
   242→   242→    } finally {
   243→   243→      setPendingId(null);
   244→   244→    }
   245→   245→  };
   246→   246→
   247→   247→  const toggleActive = async (p: Participant) => {
   248→   248→    try {
   249→   249→      await api<{ participant: Participant }>(`/api/participants/${p.id}`, {
   250→   250→        method: "PATCH",
   251→   251→        body: JSON.stringify({ active: !p.active }),
   252→   252→      });
   253→   253→      toast.success(`${p.name} ${p.active ? "deactivated" : "reactivated"}`);
   254→   254→      onChanged();
   255→   255→    } catch (err) {
   256→   256→      toast.error(apiErrorMessage(err));
   257→   257→    }
   258→   258→  };
   259→   259→
   260→   260→  return (
   261→   261→    <Card className="rounded-xl p-4 sm:p-6">
   262→   262→      <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   263→   263→        <div>
   264→   264→          <h2 className="text-sm font-semibold">Participant consent</h2>
   265→   265→          <p className="text-xs text-muted-foreground">
   266→   266→            Enrollment requires consent. Withdrawing consent removes the participant from future campaigns.
   267→   267→          </p>
   268→   268→        </div>
   269→   269→        <Button size="sm" onClick={() => setAddOpen(true)}>
   270→   270→          <UserPlus className="size-4" aria-hidden="true" />
   271→   271→          Add participant
   272→   272→        </Button>
   273→   273→      </div>
   274→   274→
   275→   275→      {consent.length === 0 ? (
   276→   276→        <EmptyState icon={UserPlus} title="No participants yet" hint="Add your first participant to get started." />
   277→   277→      ) : (
   278→   278→        <div className="overflow-x-auto">
   279→   279→          <Table>
   280→   280→            <TableHeader>
   281→   281→              <TableRow>
   282→   282→                <TableHead>Name</TableHead>
   283→   283→                <TableHead>Email</TableHead>
   284→   284→                <TableHead>Department</TableHead>
   285→   285→                <TableHead>Consent</TableHead>
   286→   286→                <TableHead>Recorded</TableHead>
   287→   287→                <TableHead className="text-right">Actions</TableHead>
   288→   288→              </TableRow>
   289→   289→            </TableHeader>
   290→   290→            <TableBody>
   291→   291→              {consent.map((row) => {
   292→   292→                const full = participantById.get(row.id);
   293→   293→                return (
   294→   294→                  <TableRow key={row.id} className={full && !full.active ? "opacity-60" : ""}>
   295→   295→                    <TableCell>
   296→   296→                      <div className="text-sm font-medium">{row.name}</div>
   297→   297→                      {full && !full.active ? (
   298→   298→                        <Badge variant="outline" className="mt-1 text-[10px] text-muted-foreground">
   299→   299→                          deactivated
   300→   300→                        </Badge>
   301→   301→                      ) : null}
   302→   302→                    </TableCell>
   303→   303→                    <TableCell className="font-mono text-xs">{row.email}</TableCell>
   304→   304→                    <TableCell className="text-xs">{row.department}</TableCell>
   305→   305→                    <TableCell>
   306→   306→                      <Switch
   307→   307→                        checked={row.consented}
   308→   308→                        disabled={pendingId === row.id}
   309→   309→                        onCheckedChange={(v) => void setConsent(row.id, v)}
   310→   310→                        aria-label={`Consent for ${row.name}`}
   311→   311→                      />
   312→   312→                    </TableCell>
   313→   313→                    <TableCell className="text-xs text-muted-foreground">
   314→   314→                      {row.consented && row.consentedAt
   315→   315→                        ? format(new Date(row.consentedAt), "MMM d, yyyy")

   317→   317→                    </TableCell>
   318→   318→                    <TableCell className="text-right">
   319→   319→                      <DropdownMenu>

   323→   323→                          </Button>
   324→   324→                        </DropdownMenuTrigger>
   325→   325→                        <DropdownMenuContent align="end">
   326→   326→                          <DropdownMenuItem
   327→   327→                            onClick={() => full && void toggleActive(full)}
   328→   328→                            disabled={!full}
   329→   329→                          >
   330→   330→                            {full?.active ? "Deactivate participant" : "Reactivate participant"}
   331→   331→                          </DropdownMenuItem>
   332→   332→                        </DropdownMenuContent>
   333→   333→                      </DropdownMenu>
   334→   334→                    </TableCell>
   335→   335→                  </TableRow>
   336→   336→                );
   337→   337→              })}
   338→   338→            </TableBody>
   339→   339→          </Table>
   340→   340→        </div>
   341→   341→      )}
   342→   342→
   343→   343→      {addOpen ? (
   344→   344→        <AddParticipantDialog
   345→   345→          open={addOpen}
   346→   346→          onClose={() => setAddOpen(false)}
   347→   347→          onSaved={onChanged}
   348→   348→        />
   349→   349→      ) : null}
   350→   350→    </Card>
   351→   351→  );
   352→   352→}

   450→   450→function DomainsTab({
   451→   451→  domains,
   452→   452→  onChanged,
   453→   453→}: {
   454→   454→  domains: { id: string; domain: string; approvedBy: string; createdAt: string }[];
   455→   455→  onChanged: () => void;
   456→   456→}) {
   457→   457→  const [domain, setDomain] = useState("");
   458→   458→  const [approvedBy, setApprovedBy] = useState("security.admin@acme-corp.test");
   459→   459→  const [busy, setBusy] = useState(false);
   460→   460→
   461→   461→  const addDomain = async () => {
   462→   462→    const d = domain.trim().toLowerCase();
   463→   463→    if (!d || !d.includes(".")) {
   464→   464→      toast.error("Enter a valid domain, e.g. portal.acme-corp.test");
   465→   465→      return;
   466→   466→    }
   467→   467→    setBusy(true);
   468→   468→    try {
   469→   469→      await api<{ domain: unknown }>("/api/compliance/domains", {
   470→   470→        method: "POST",
   471→   471→        body: JSON.stringify({ domain: d, approvedBy: approvedBy.trim() }),
   472→   472→      });
   473→   473→      toast.success(`Domain ${d} approved`);
   474→   474→      setDomain("");
   475→   475→      onChanged();
   476→   476→    } catch (err) {
   477→   477→      toast.error(apiErrorMessage(err));
   478→   478→    } finally {

   580→   580→
   581→   581→function RetentionTab({
   582→   582→  retention,
   583→   583→  onChanged,
   584→   584→}: {
   585→   585→  retention: {
   586→   586→    defaultRetentionDays: number;
   587→   587→    autoPurge: boolean;
   588→   588→    totalEvents: number;
   589→   589→    oldestEventAt: string | null;
   590→   590→  };
   591→   591→  onChanged: () => void;
   592→   592→}) {
   593→   593→  const [days, setDays] = useState(String(retention.defaultRetentionDays));
   594→   594→  const [autoPurge, setAutoPurge] = useState(retention.autoPurge);
   595→   595→  const [saving, setSaving] = useState(false);
   596→   596→
   597→   597→  const oldestAgeDays = retention.oldestEventAt
   598→   598→    ? Math.max(
   599→   599→        0,
   600→   600→        Math.floor((Date.now() - new Date(retention.oldestEventAt).getTime()) / 86_400_000)
   601→   601→      )
   602→   602→    : null;
   603→   603→
   604→   604→  const save = async () => {
   605→   605→    const n = Number.parseInt(days, 10);
   606→   606→    if (!Number.isFinite(n) || n < 1 || n > 365) {
   607→   607→      toast.error("Retention must be between 1 and 365 days.");
   608→   608→      return;
   609→   609→    }
   610→   610→    setSaving(true);
   611→   611→    try {
   612→   612→      await api<{ retention: unknown }>("/api/compliance/retention", {
   613→   613→        method: "PATCH",
   614→   614→        body: JSON.stringify({ defaultRetentionDays: n, autoPurge }),
   615→   615→      });
   616→   616→      toast.success("Retention policy saved");
   617→   617→      onChanged();
   618→   618→    } catch (err) {
   619→   619→      toast.error(apiErrorMessage(err));
   620→   620→    } finally {
   621→   621→      setSaving(false);
   622→   622→    }
   623→   623→  };
   624→   624→
   625→   625→  return (
   626→   626→    <div className="grid gap-4 lg:grid-cols-2">
   627→   627→      <Card className="rounded-xl p-4 sm:p-6">
   628→   628→        <h2 className="text-sm font-semibold">Retention policy</h2>
   629→   629→        <p className="mt-1 text-xs text-muted-foreground">
   630→   630→          Applies to simulated events (deliveries, opens, clicks, reports) across all campaigns.
   631→   631→        </p>
   632→   632→        <div className="mt-4 space-y-4">
   633→   633→          <div className="space-y-1.5">
   634→   634→            <Label htmlFor="ret-days">Default retention (days)</Label>
   635→   635→            <Input
   636→   636→              id="ret-days"
   637→   637→              type="number"
   638→   638→              min={1}
   639→   639→              max={365}
   640→   640→              value={days}
   641→   641→              onChange={(e) => setDays(e.target.value)}
   642→   642→              className="w-40"
   643→   643→            />
   644→   644→          </div>
   645→   645→          <div className="flex items-center gap-3">
   646→   646→            <Switch
   647→   647→              id="ret-purge"
   648→   648→              checked={autoPurge}
   649→   649→              onCheckedChange={setAutoPurge}
   650→   650→              aria-label="Auto purge"
   651→   651→            />
   652→   652→            <Label htmlFor="ret-purge" className="text-sm font-normal">
   653→   653→              Auto-purge events after the retention window
   654→   654→            </Label>
   655→   655→          </div>
   656→   656→          <Button onClick={() => void save()} disabled={saving}>
   657→   657→            {saving ? "Saving…" : "Save policy"}
   658→   658→          </Button>
   659→   659→        </div>
   660→   660→      </Card>
   661→   661→
   662→   662→      <div className="space-y-4">
   663→   663→        <Card className="rounded-xl p-4 sm:p-6">
   664→   664→          <h2 className="text-sm font-semibold">Current data footprint</h2>
   665→   665→          <dl className="mt-3 space-y-2 text-sm">
   666→   666→            <div className="flex justify-between">
   667→   667→              <dt className="text-muted-foreground">Tracked simulated events</dt>
   668→   668→              <dd className="font-semibold tabular-nums">{retention.totalEvents}</dd>
   669→   669→            </div>
   670→   670→            <div className="flex justify-between">
   671→   671→              <dt className="text-muted-foreground">Oldest event age</dt>
   672→   672→              <dd className="font-semibold tabular-nums">
   673→   673→                {oldestAgeDays === null ? "no events yet" : `${oldestAgeDays} days`}
   674→   674→              </dd>
   675→   675→            </div>
   676→   676→            <div className="flex justify-between">
   677→   677→              <dt className="text-muted-foreground">Auto-purge</dt>
   678→   678→              <dd className={autoPurge ? "font-semibold text-emerald-400" : "font-semibold text-amber-400"}>
   679→   679→                {autoPurge ? "enabled" : "disabled"}
   680→   680→              </dd>
   681→   681→            </div>
   682→   682→          </dl>
   683→   683→        </Card>
   684→   684→        <div
   685→   685→          role="note"
   686→   686→          className="flex items-start gap-3 rounded-xl border border-red-500/30 bg-red-500/10 p-4"
   687→   687→        >
   688→   688→          <ShieldCheck className="mt-0.5 size-4 shrink-0 text-red-400" aria-hidden="true" />
   689→   689→          <p className="text-xs leading-relaxed text-red-200/90">
   690→   690→            Events older than the retention window are permanently purged. This cannot be undone —
   691→   691→            exports should be taken before reducing the window.
   692→   692→          </p>
   693→   693→        </div>
   694→   694→      </div>
   695→   695→    </div>
   696→   696→  );
   697→   697→}
