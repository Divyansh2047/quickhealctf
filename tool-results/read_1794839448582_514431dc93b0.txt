    96→function auditBadgeClass(action: string): string {
    97→  if (action === "CAMPAIGN_LAUNCHED" || action === "CONSENT_RECORDED")
    98→    return "border-emerald-500/40 bg-emerald-500/15 text-emerald-400";
    99→  if (action === "DOMAIN_APPROVED")
   100→    return "border-teal-500/40 bg-teal-500/15 text-teal-400";
   101→  if (action === "TEMPLATE_CREATED" || action === "RETENTION_POLICY")
   102→    return "border-amber-500/40 bg-amber-500/15 text-amber-400";
   103→  return "border-zinc-500/40 bg-zinc-500/15 text-zinc-400";
   104→}

   217→/* ------------------------------ consent tab ------------------------------ */
   218→
   219→function ConsentTab({
   220→  consent,
   221→  participantById,
   222→  onChanged,
   223→}: {
   224→  consent: { id: string; name: string; email: string; department: string; consented: boolean; consentedAt: string | null }[];
   225→  participantById: Map<string, Participant>;
   226→  onChanged: () => void;
   227→}) {
   228→  const [addOpen, setAddOpen] = useState(false);
   229→  const [pendingId, setPendingId] = useState<string | null>(null);
   230→
   231→  const setConsent = async (id: string, consented: boolean) => {
   232→    setPendingId(id);
   233→    try {
   234→      await api<{ participant: Participant }>(`/api/participants/${id}`, {
   235→        method: "PATCH",
   236→        body: JSON.stringify({ consented }),
   237→      });
   238→      toast.success(consented ? "Consent recorded" : "Consent withdrawn");
   239→      onChanged();
   240→    } catch (err) {
   241→      toast.error(apiErrorMessage(err));
   242→    } finally {
   243→      setPendingId(null);
   244→    }
   245→  };
   246→
   247→  const toggleActive = async (p: Participant) => {
   248→    try {
   249→      await api<{ participant: Participant }>(`/api/participants/${p.id}`, {
   250→        method: "PATCH",
   251→        body: JSON.stringify({ active: !p.active }),
   252→      });
   253→      toast.success(`${p.name} ${p.active ? "deactivated" : "reactivated"}`);
   254→      onChanged();
   255→    } catch (err) {
   256→      toast.error(apiErrorMessage(err));
   257→    }
   258→  };
   259→
   260→  return (
   261→    <Card className="rounded-xl p-4 sm:p-6">
   262→      <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   263→        <div>
   264→          <h2 className="text-sm font-semibold">Participant consent</h2>
   265→          <p className="text-xs text-muted-foreground">
   266→            Enrollment requires consent. Withdrawing consent removes the participant from future campaigns.
   267→          </p>
   268→        </div>
   269→        <Button size="sm" onClick={() => setAddOpen(true)}>
   270→          <UserPlus className="size-4" aria-hidden="true" />
   271→          Add participant
   272→        </Button>
   273→      </div>
   274→
   275→      {consent.length === 0 ? (
   276→        <EmptyState icon={UserPlus} title="No participants yet" hint="Add your first participant to get started." />
   277→      ) : (
   278→        <div className="overflow-x-auto">
   279→          <Table>
   280→            <TableHeader>
   281→              <TableRow>
   282→                <TableHead>Name</TableHead>
   283→                <TableHead>Email</TableHead>
   284→                <TableHead>Department</TableHead>
   285→                <TableHead>Consent</TableHead>
   286→                <TableHead>Recorded</TableHead>
   287→                <TableHead className="text-right">Actions</TableHead>
   288→              </TableRow>
   289→            </TableHeader>
   290→            <TableBody>
   291→              {consent.map((row) => {
   292→                const full = participantById.get(row.id);
   293→                return (
   294→                  <TableRow key={row.id} className={full && !full.active ? "opacity-60" : ""}>
   295→                    <TableCell>
   296→                      <div className="text-sm font-medium">{row.name}</div>
   297→                      {full && !full.active ? (
   298→                        <Badge variant="outline" className="mt-1 text-[10px] text-muted-foreground">
   299→                          deactivated
   300→                        </Badge>
   301→                      ) : null}
   302→                    </TableCell>
   303→                    <TableCell className="font-mono text-xs">{row.email}</TableCell>
   304→                    <TableCell className="text-xs">{row.department}</TableCell>
   305→                    <TableCell>
   306→                      <Switch
   307→                        checked={row.consented}
   308→                        disabled={pendingId === row.id}
   309→                        onCheckedChange={(v) => void setConsent(row.id, v)}
   310→                        aria-label={`Consent for ${row.name}`}
   311→                      />
   312→                    </TableCell>
   313→                    <TableCell className="text-xs text-muted-foreground">
   314→                      {row.consented && row.consentedAt
   315→                        ? format(new Date(row.consentedAt), "MMM d, yyyy")
   316→                        : "—"}
   317→                    </TableCell>
   318→                    <TableCell className="text-right">
   319→                      <DropdownMenu>
   320→                        <DropdownMenuTrigger asChild>
   321→                          <Button variant="ghost" size="icon" aria-label={`Actions for ${row.name}`}>
   322→                            <MoreHorizontal className="size-4" aria-hidden="true" />
   323→                          </Button>
   324→                        </DropdownMenuTrigger>
   325→                        <DropdownMenuContent align="end">
   326→                          <DropdownMenuItem
   327→                            onClick={() => full && void toggleActive(full)}
   328→                            disabled={!full}
   329→                          >
   330→                            {full?.active ? "Deactivate participant" : "Reactivate participant"}
   331→                          </DropdownMenuItem>
   332→                        </DropdownMenuContent>
   333→                      </DropdownMenu>
   334→                    </TableCell>
   335→                  </TableRow>
   336→                );
   337→              })}
   338→            </TableBody>
   339→          </Table>
   340→        </div>
   341→      )}
   342→
   343→      {addOpen ? (
   344→        <AddParticipantDialog
   345→          open={addOpen}
   346→          onClose={() => setAddOpen(false)}
   347→          onSaved={onChanged}
   348→        />
   349→      ) : null}
   350→    </Card>
   351→  );
   352→}

   354→function AddParticipantDialog({
   355→  open,
   356→  onClose,
   357→  onSaved,
   358→}: {
   359→  open: boolean;
   360→  onClose: () => void;
   361→  onSaved: () => void;
   362→}) {
   363→  const [name, setName] = useState("");
   364→  const [email, setEmail] = useState("");
   365→  const [department, setDepartment] = useState("");
   366→  const [consented, setConsented] = useState(true);
   367→  const [saving, setSaving] = useState(false);
   368→
   369→  const submit = async () => {
   370→    if (!name.trim() || !email.trim() || !department.trim()) {
   371→      toast.error("Name, email and department are required.");
   372→      return;
   373→    }
   374→    setSaving(true);
   375→    try {
   376→      await api<{ participant: Participant }>("/api/participants", {
   377→        method: "POST",
   378→        body: JSON.stringify({
   379→          name: name.trim(),
   380→          email: email.trim(),
   381→          department: department.trim(),
   382→          consented,
   383→        }),
   384→      });
   385→      toast.success(`${name.trim()} added`);
   386→      onClose();
   387→      onSaved();
   388→    } catch (err) {
   389→      toast.error(apiErrorMessage(err));
   390→    } finally {
   391→      setSaving(false);
   392→    }
   393→  };
   394→
   395→  return (
   396→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   397→      <DialogContent className="max-w-md rounded-xl" aria-describedby="add-participant-desc">
   398→        <DialogHeader>
   399→          <DialogTitle>Add participant</DialogTitle>
   400→          <DialogDescription id="add-participant-desc">
   401→            People are enrolled in simulations only after they record consent.
   402→          </DialogDescription>
   403→        </DialogHeader>
   404→        <div className="space-y-4">
   405→          <div className="space-y-1.5">
   406→            <Label htmlFor="p-name">Full name</Label>
   407→            <Input id="p-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Jordan Lee" />
   408→          </div>
   409→          <div className="space-y-1.5">
   410→            <Label htmlFor="p-email">Work email</Label>
   411→            <Input
   412→              id="p-email"
   413→              type="email"
   414→              value={email}
   415→              onChange={(e) => setEmail(e.target.value)}
   416→              placeholder="jordan.lee@acme-corp.test"
   417→            />
   418→          </div>
   419→          <div className="space-y-1.5">
   420→            <Label htmlFor="p-dept">Department</Label>
   421→            <Input id="p-dept" value={department} onChange={(e) => setDepartment(e.target.value)} placeholder="Finance" />
   422→          </div>
   423→          <div className="flex items-center gap-2">
   424→            <Checkbox
   425→              id="p-consent"
   426→              checked={consented}
   427→              onCheckedChange={(v) => setConsented(v === true)}
   428→              aria-label="Consent recorded now"
   429→            />
   430→            <Label htmlFor="p-consent" className="text-sm font-normal">
   431→              Consent recorded now (participant signed the authorization form)
   432→            </Label>
   433→          </div>
   434→        </div>
   435→        <DialogFooter className="border-t pt-4">
   436→          <Button variant="ghost" onClick={onClose} disabled={saving}>
   437→            Cancel
   438→          </Button>
   439→          <Button onClick={() => void submit()} disabled={saving}>
   440→            {saving ? "Adding…" : "Add participant"}
   441→          </Button>
   442→        </DialogFooter>
   443→      </DialogContent>
   444→    </Dialog>
   445→  );
   446→}

   450→function DomainsTab({
   451→  domains,
   452→  onChanged,
   453→}: {
   454→  domains: { id: string; domain: string; approvedBy: string; createdAt: string }[];
   455→  onChanged: () => void;
   456→}) {
   457→  const [domain, setDomain] = useState("");
   458→  const [approvedBy, setApprovedBy] = useState("security.admin@acme-corp.test");
   459→  const [busy, setBusy] = useState(false);
   460→
   461→  const addDomain = async () => {
   462→    const d = domain.trim().toLowerCase();
   463→    if (!d || !d.includes(".")) {
   464→      toast.error("Enter a valid domain, e.g. portal.acme-corp.test");
   465→      return;
   466→    }
   467→    setBusy(true);
   468→    try {
   469→      await api<{ domain: unknown }>("/api/compliance/domains", {
   470→        method: "POST",
   471→        body: JSON.stringify({ domain: d, approvedBy: approvedBy.trim() }),
   472→      });
   473→      toast.success(`Domain ${d} approved`);
   474→      setDomain("");
   475→      onChanged();
   476→    } catch (err) {
   477→      toast.error(apiErrorMessage(err));
   478→    } finally {
   479→      setBusy(false);
   480→    }
   481→  };
   482→
   483→  const revoke = async (id: string, d: string) => {
   484→    try {
   485→      await api<{ ok: boolean }>(`/api/compliance/domains/${id}`, { method: "DELETE" });
   486→      toast.success(`Domain ${d} revoked`);
   487→      onChanged();
   488→    } catch (err) {
   489→      toast.error(apiErrorMessage(err));
   490→    }
   491→  };
   492→
   493→  return (
   494→    <div className="grid gap-4 lg:grid-cols-[1fr_320px]">
   495→      <Card className="rounded-xl p-4 sm:p-6">
   496→        <h2 className="text-sm font-semibold">Approved sender domains</h2>
   497→        <p className="mt-1 text-xs text-muted-foreground">
   498→          Campaign launches are rejected when the template sender domain is not on this allowlist.
   499→        </p>
   500→        {domains.length === 0 ? (
   501→          <EmptyState icon={Shield} title="No approved domains" hint="Approve a sender domain to allow launches." className="mt-3" />
   502→        ) : (
   503→          <ul className="mt-3 max-h-96 space-y-2 overflow-y-auto pr-1">
   504→            {domains.map((d) => (
   505→              <li
   506→                key={d.id}
   507→                className="flex flex-wrap items-center justify-between gap-2 rounded-lg border p-3"
   508→              >
   509→                <div className="min-w-0">
   510→                  <p className="font-mono text-sm font-medium">{d.domain}</p>
   511→                  <p className="mt-0.5 text-xs text-muted-foreground">
   512→                    approved by <span className="font-mono">{d.approvedBy}</span> ·{" "}
   513→                    {format(new Date(d.createdAt), "MMM d, yyyy")}
   514→                  </p>
   515→                </div>
   516→                <AlertDialog>
   517→                  <AlertDialogTrigger asChild>
   518→                    <Button variant="outline" size="sm" className="text-red-400 hover:text-red-300">
   519→                      Revoke
   520→                    </Button>
   521→                  </AlertDialogTrigger>
   522→                  <AlertDialogContent>
   523→                    <AlertDialogHeader>
   524→                      <AlertDialogTitle>Revoke {d.domain}?</AlertDialogTitle>
   525→                      <AlertDialogDescription>
   526→                        Campaigns using templates from this domain can no longer be launched.
   527→                        Existing campaigns are unaffected.
   528→                      </AlertDialogDescription>
   529→                    </AlertDialogHeader>
   530→                    <AlertDialogFooter>
   531→                      <AlertDialogCancel>Keep</AlertDialogCancel>
   532→                      <AlertDialogAction
   533→                        className="bg-destructive text-white hover:bg-destructive/90"
   534→                        onClick={() => void revoke(d.id, d.domain)}
   535→                      >
   536→                        Revoke domain
   537→                      </AlertDialogAction>
   538→                    </AlertDialogFooter>
   539→                  </AlertDialogContent>
   540→                </AlertDialog>
   541→              </li>
   542→            ))}
   543→          </ul>
   544→        )}
   545→      </Card>
   546→
   547→      <Card className="h-fit rounded-xl p-4 sm:p-6">
   548→        <h2 className="text-sm font-semibold">Approve a domain</h2>
   549→        <div className="mt-3 space-y-3">
   550→          <div className="space-y-1.5">
   551→            <Label htmlFor="dom-domain">Domain</Label>
   552→            <Input
   553→              id="dom-domain"
   554→              value={domain}
   555→              onChange={(e) => setDomain(e.target.value)}
   556→              placeholder="portal.acme-corp.test"
   557→              className="font-mono text-sm"
   558→            />
   559→          </div>
   560→          <div className="space-y-1.5">
   561→            <Label htmlFor="dom-approver">Approved by</Label>
   562→            <Input
   563→              id="dom-approver"
   564→              value={approvedBy}
   565→              onChange={(e) => setApprovedBy(e.target.value)}
   566→              className="font-mono text-sm"
   567→            />
   568→          </div>
   569→          <Button className="w-full" onClick={() => void addDomain()} disabled={busy}>
   570→            <Plus className="size-4" aria-hidden="true" />
   571→            {busy ? "Approving…" : "Approve domain"}
   572→          </Button>
   573→        </div>

   579→/* ----------------------------- retention tab ----------------------------- */
   580→
   581→function RetentionTab({
   582→  retention,
   583→  onChanged,
   584→}: {
   585→  retention: {
   586→    defaultRetentionDays: number;
   587→    autoPurge: boolean;
   588→    totalEvents: number;
   589→    oldestEventAt: string | null;
   590→  };
   591→  onChanged: () => void;
   592→}) {
   593→  const [days, setDays] = useState(String(retention.defaultRetentionDays));
   594→  const [autoPurge, setAutoPurge] = useState(retention.autoPurge);
   595→  const [saving, setSaving] = useState(false);
   596→
   597→  const oldestAgeDays = retention.oldestEventAt
   598→    ? Math.max(
   599→        0,
   600→        Math.floor((Date.now() - new Date(retention.oldestEventAt).getTime()) / 86_400_000)
   601→      )
   602→    : null;
   603→
   604→  const save = async () => {
   605→    const n = Number.parseInt(days, 10);
   606→    if (!Number.isFinite(n) || n < 1 || n > 365) {
   607→      toast.error("Retention must be between 1 and 365 days.");
   608→      return;
   609→    }
   610→    setSaving(true);
   611→    try {
   612→      await api<{ retention: unknown }>("/api/compliance/retention", {
   613→        method: "PATCH",
   614→        body: JSON.stringify({ defaultRetentionDays: n, autoPurge }),
   615→      });
   616→      toast.success("Retention policy saved");
   617→      onChanged();
   618→    } catch (err) {
   619→      toast.error(apiErrorMessage(err));
   620→    } finally {
   621→      setSaving(false);
   622→    }
   623→  };
   624→
   625→  return (
   626→    <div className="grid gap-4 lg:grid-cols-2">
   627→      <Card className="rounded-xl p-4 sm:p-6">
   628→        <h2 className="text-sm font-semibold">Retention policy</h2>
   629→        <p className="mt-1 text-xs text-muted-foreground">
   630→          Applies to simulated events (deliveries, opens, clicks, reports) across all campaigns.
   631→        </p>
   632→        <div className="mt-4 space-y-4">
   633→          <div className="space-y-1.5">
   634→            <Label htmlFor="ret-days">Default retention (days)</Label>
   635→            <Input
   636→              id="ret-days"
   637→              type="number"
   638→              min={1}
   639→              max={365}
   640→              value={days}
   641→              onChange={(e) => setDays(e.target.value)}
   642→              className="w-40"
   643→            />
   644→          </div>
   645→          <div className="flex items-center gap-3">
   646→            <Switch
   647→              id="ret-purge"
   648→              checked={autoPurge}
   649→              onCheckedChange={setAutoPurge}
   650→              aria-label="Auto purge"
   651→            />
   652→            <Label htmlFor="ret-purge" className="text-sm font-normal">
   653→              Auto-purge events after the retention window
   654→            </Label>
   655→          </div>
   656→          <Button onClick={() => void save()} disabled={saving}>
   657→            {saving ? "Saving…" : "Save policy"}
   658→          </Button>
   659→        </div>
   660→      </Card>
   661→
   662→      <div className="space-y-4">
   663→        <Card className="rounded-xl p-4 sm:p-6">
   664→          <h2 className="text-sm font-semibold">Current data footprint</h2>
   665→          <dl className="mt-3 space-y-2 text-sm">
   666→            <div className="flex justify-between">
   667→              <dt className="text-muted-foreground">Tracked simulated events</dt>
   668→              <dd className="font-semibold tabular-nums">{retention.totalEvents}</dd>
   669→            </div>
   670→            <div className="flex justify-between">
   671→              <dt className="text-muted-foreground">Oldest event age</dt>
   672→              <dd className="font-semibold tabular-nums">
   673→                {oldestAgeDays === null ? "no events yet" : `${oldestAgeDays} days`}
   674→              </dd>
   675→            </div>
   676→            <div className="flex justify-between">
   677→              <dt className="text-muted-foreground">Auto-purge</dt>
   678→              <dd className={autoPurge ? "font-semibold text-emerald-400" : "font-semibold text-amber-400"}>
   679→                {autoPurge ? "enabled" : "disabled"}
   680→              </dd>
   681→            </div>
   682→          </dl>
   683→        </Card>
   684→        <div
   685→          role="note"
   686→          className="flex items-start gap-3 rounded-xl border border-red-500/30 bg-red-500/10 p-4"
   687→        >
   688→          <ShieldCheck className="mt-0.5 size-4 shrink-0 text-red-400" aria-hidden="true" />
   689→          <p className="text-xs leading-relaxed text-red-200/90">
   690→            Events older than the retention window are permanently purged. This cannot be undone —
   691→            exports should be taken before reducing the window.
   692→          </p>
   693→        </div>
   694→      </div>
   695→    </div>
   696→  );
   697→}
